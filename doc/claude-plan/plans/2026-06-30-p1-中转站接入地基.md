# P1 中转站接入地基 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: 用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐 Task 实施。步骤用 `- [ ]` 复选框跟踪。

**Goal:** 让客户端登录 Claude360 中转站（newapi）、自动获取调用凭据、把对话/写作运行时与模型下拉统一收敛到中转站，用户无需配置任何模型供应商。

**Architecture:** 新增主进程「Claude360 网关代理模块」作为客户端到 newapi 的唯一出口（注入 access token、New-Api-Group 分组头，经现有 `proxy-fetch` 支持系统代理）；渲染进程经新 IPC 调用，绝不直连（CSP `default-src 'self'` 强制）。凭据持久化进现有 `JsonSettingsStore` 的新 `account` 段。登录弹窗替换首启配置弹窗，双入口（设备码 / 用户名密码）归一到 access token；登录后自动确保「Copilot 专属 Key」并注入 `kun serve` 运行时。运行时收敛为单一中转站 provider；模型下拉来源改为 `GET /api/cli/models`。

**Tech Stack:** Electron 34（主进程 Node + IPC）+ React 19 + TS（双 tsconfig）+ Vitest + zod（IPC 校验）+ zustand（renderer store）。后端 newapi（Go/Gin，接口只读、不改）。

**前置依赖：** 无硬性前置（可在 M1 之后、与 P0/P2 解耦执行）。但本计划新增的「我的」页所需账户接口在 P3 复用本模块，故 P1 必须先行。

**已核实的后端契约（newapi，只读不改）：** `/api/cli/*` 全部存在于 `controller/cli.go`；鉴权头 `Authorization`（带不带 `Bearer ` 均可，中间件 `CliAccessTokenAuth`）；分组头 `New-Api-Group`（`middleware/distributor.go:87`，仅校验"在用户可用分组"）；换算常量 `QuotaPerUnit=500000`、`Price=7.3`；`/api/status` 返回 `quota_per_unit`/`price` 但**不返回 password 登录开关**。详见下文各 Task 的「契约」块。

`src/shared/app-settings-account.ts`：
```ts
export type AccountSettingsV1 = {
  accessToken: string      // newapi user.access_token（身份令牌）
  copilotApiKey: string    // Copilot 专属 sk-xxx（调用凭据）
  baseUrl: string          // 覆盖中转站地址，空=用 RELAY_BASE_URL（K2）
  username: string
  displayName: string
}

export function defaultAccountSettings(): AccountSettingsV1 {
  return { accessToken: '', copilotApiKey: '', baseUrl: '', username: '', displayName: '' }
}

export function normalizeAccountSettings(input: Partial<AccountSettingsV1> | undefined): AccountSettingsV1 {
  const base = defaultAccountSettings()
  if (!input || typeof input !== 'object') return base
  const pick = (v: unknown): string => (typeof v === 'string' ? v.trim() : '')
  return {
    accessToken: pick(input.accessToken),
    copilotApiKey: pick(input.copilotApiKey),
    baseUrl: pick(input.baseUrl),
    username: pick(input.username),
    displayName: pick(input.displayName)
  }
}

export function mergeAccountSettings(
  cur: AccountSettingsV1,
  patch: Partial<AccountSettingsV1> | undefined
): AccountSettingsV1 {
  if (!patch) return cur
  return normalizeAccountSettings({ ...cur, ...patch })
}
```

- [ ] **Step 2：实现网关模块**

`src/main/relay/relay-gateway.ts`：
```ts
import { fetchWithOptionalProxy } from '../proxy-fetch'
import type { AccountSettingsV1 } from '@shared/app-settings-account'

export type RelayRequest = {
  path: string
  method: 'GET' | 'POST' | 'DELETE'
  auth?: boolean
  group?: string
  body?: unknown
  /** 用 cookie 而非 access token（仅 /api/user/token 等控制台接口）。 */
  cookie?: string
}

export class RelayError extends Error {
  constructor(public status: number, message: string, public body?: unknown) { super(message) }
}

export type RelayGatewayDeps = {
  getAccount: () => AccountSettingsV1
  getProxyUrl: () => string
  baseUrlFallback: string
  fetchImpl?: typeof fetch
}

export function createRelayGateway(deps: RelayGatewayDeps) {
  const doFetch = deps.fetchImpl ?? ((u: string, i?: RequestInit) => fetchWithOptionalProxy(u, i, deps.getProxyUrl()))
  async function request<T = unknown>(req: RelayRequest): Promise<T> {
    const account = deps.getAccount()
    const base = (account.baseUrl || deps.baseUrlFallback).replace(/\/+$/, '')
    const headers: Record<string, string> = { Accept: 'application/json' }
    if (req.body !== undefined) headers['Content-Type'] = 'application/json'
    if (req.auth) headers.Authorization = account.accessToken
    if (req.cookie) headers.Cookie = req.cookie
    if (req.group) headers['New-Api-Group'] = req.group
    const res = await doFetch(base + req.path, {
      method: req.method,
      headers,
      body: req.body !== undefined ? JSON.stringify(req.body) : undefined
    })
    const text = await res.text()
    let parsed: unknown = undefined
    try { parsed = text ? JSON.parse(text) : undefined } catch { parsed = text }
    if (!res.ok) throw new RelayError(res.status, `relay ${req.method} ${req.path} -> ${res.status}`, parsed)
    // newapi 控制台风格响应常为 { success, message, data }；CLI 接口直接返回数据。两种都透传。
    return parsed as T
  }
  return { request }
}
export type RelayGateway = ReturnType<typeof createRelayGateway>
```

```ts
import type { RelayGateway } from './relay-gateway'
import { COPILOT_TOKEN_NAME } from '@shared/relay-constants'

type TokenItem = { id: number | string; name: string }

export async function ensureCopilotApiKey(gateway: RelayGateway): Promise<string> {
  const list = await gateway.request<{ items?: TokenItem[] }>({ path: '/api/cli/tokens', method: 'GET', auth: true })
  const existing = (list.items ?? []).find((t) => t.name === COPILOT_TOKEN_NAME)
  if (existing) {
    const revealed = await gateway.request<{ key: string }>({ path: `/api/cli/tokens/${existing.id}/reveal`, method: 'POST', auth: true })
    return revealed.key
  }
  const created = await gateway.request<{ key: string }>({ path: '/api/cli/tokens', method: 'POST', auth: true, body: { name: COPILOT_TOKEN_NAME } })
  return created.key
}
```

// 设备码：start → 打开浏览器 → 轮询直到 approved/denied/expired
export async function startDeviceLogin(set, get): Promise<void> {
  const s = await window.kunGui.relay.authStart()
  await window.kunGui.openExternal(s.verification_url)  // 见下注：openExternal IPC
  set({ deviceUserCode: s.user_code })
  const deadline = Date.now() + s.expires_in * 1000
  const tick = async () => {
    if (Date.now() > deadline) { set({ loginError: 'expired' }); return }
    const r = await window.kunGui.relay.authPoll(s.device_code)
    if (r.status === 'approved') return finishLogin(r.cli_token, set, get)
    if (r.status === 'denied' || r.status === 'expired') { set({ loginError: r.status }); return }
    setTimeout(tick, s.interval * 1000)
  }
  setTimeout(tick, s.interval * 1000)
}

// 用户名密码：login → (2fa?) → /api/user/token 取 access token
export async function passwordLogin(username, password, set, get): Promise<void> {
  const res = await window.kunGui.relay.login(username, password)
  const cookie = res.__setCookie ?? ''
  if (res.data?.require_2fa) { set({ pending2fa: { cookie } }); return }
  const token = await window.kunGui.relay.generateToken(cookie)
  return finishLogin(token.data ?? token, set, get)
}

// 归一收尾：写 account + 确保 Copilot Key + 注入运行时 + 关弹窗 + 探测运行时
async function finishLogin(accessToken: string, set, get): Promise<void> {
  await window.kunGui.saveSettingsSilent({ account: { accessToken } })
  const apiKey = await window.kunGui.relay.ensureCopilotKey()   // 见注：新增 ensure IPC
  const me = await window.kunGui.relay.me()
  await window.kunGui.saveSettingsSilent({
    account: { accessToken, copilotApiKey: apiKey, username: me.username, displayName: me.display_name }
  })
  // 注入运行时 provider（Task 7 收敛后，运行时从 account.copilotApiKey 取 key）
  await get().reloadUiSettings()
  set({ loginOpen: false, loginError: null })
  await get().probeRuntime()
}
